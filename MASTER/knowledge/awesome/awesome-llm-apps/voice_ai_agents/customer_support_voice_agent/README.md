# 1️⃣ Clone the tutorial repository

cd awesome-llm-apps/ai_agent_tutorials/ai_voice_agent_openaisdk

# 2️⃣ Create an isolated virtual environment

.venv\Scripts\Activate.ps1

pip install -r requirements.txt
